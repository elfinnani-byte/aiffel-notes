# [에이전트 팀 꾸리기] - 4. 실습 ③ 라우터 그래프를 세우고 규칙 노드를 꽂는다

## 먼저 실행해 봅시다 — 그래프를 세우고 규칙 노드를 꽂는다

오늘 만들 라우터는 함수 하나가 아니라 **작은 그래프**입니다. 문의가 들어오면 ① 어느 라우트인지 분류하고 ② 그 판단을 믿고 넘길지 사람에게 이관할지 결정합니다. 두 단계는 성격이 다릅니다. 분류는 모델이 하는 일이고, 판정은 회사 정책이 정하는 일이거든요. 섞어 두면 나중에 모델만 바꾸고 싶을 때 정책까지 건드리게 됩니다. 그래서 **노드 두 개로 나눠** 세웁니다.

그래프를 먼저 세우고 그 안을 채우는 순서에는 이유가 있어요. 오늘 우리는 분류 노드를 **규칙 → LLM → LLM+예시**로 세 번 갈아 끼웁니다. 그때마다 그래프를 다시 그리지 않고 **같은 이름의 함수를 다시 정의하기만** 하면 됩니다. `build()` 가 노드를 이름으로 찾아가거든요. 바뀐 것은 노드 하나뿐이고 나머지는 그대로이므로, 점수가 오르면 그 원인이 무엇인지 헷갈릴 일이 없습니다.

첫 번째 분류 노드는 **키워드 규칙**입니다. "환불"이 있으면 `RETURN_REFUND`, "배송비"가 있으면 `SHIPPING`. 이런 걸 왜 만드느냐면, **LLM 없이 얼마나 되는지**를 알아야 LLM이 정말 필요한지 판단할 수 있기 때문이에요. 규칙만으로 95%가 나온다면 API 비용을 쓸 이유가 없습니다. 그리고 규칙이 실패하는 지점이 곧 **문제의 본질**을 드러냅니다. 어디서 왜 실패하는지 보면 LLM에게 무엇을 시켜야 하는지가 분명해집니다.

➡️**실습: 라우터 그래프를 세우고 규칙 분류 노드를 꽂기**
```python
import re
from typing import TypedDict, Optional
from langgraph.graph import StateGraph, START, END

def show_graph(compiled):
    """그래프 구조를 mermaid 로 그려 본다. LMS와 Jupyter 모두 mermaid 를 렌더링한다."""
    src = compiled.get_graph().draw_mermaid()
    try:
        from IPython.display import Markdown, display
        display(Markdown("```mermaid\n" + src + "\n```"))
    except ImportError:
        print(src)


class RouterState(TypedDict, total=False):
    """그래프를 통과하며 채워지는 값들. 노드는 자기가 바꾼 키만 돌려준다."""

    question: str          # 입력 — 고객 문의 한 문장
    route: str             # ① 분류 노드가 채운다
    confidence: float      # ① 분류 노드가 채운다
    reason: str            # ① 분류 노드가 채운다
    action: str            # ② 판정 노드가 채운다 — HANDLE / ESCALATE / OUT_OF_SCOPE
    message: Optional[str]  # ② 판정 노드가 채운다 — 고객에게 바로 나갈 문구


# 라우트별 키워드 규칙. 위에서부터 먼저 매치되는 규칙을 채택한다(순서가 곧 우선순위).
RULES = [
    (r"환불|반품|교환|취소|반송|수거|검품", "RETURN_REFUND"),
    (r"배송비|택배비|무료\s?배송|배송|택배|출고|도착|언제\s?(오|와)|어디쯤", "SHIPPING"),
    (r"주문할|구매할게|살게요|사고\s?싶|결제|주문\s?가능|구매\s?가능|낱개|따로|단품", "ORDER_PLACE"),
    (r"구성|포함|소재|재질|사이즈|치수|재고|정품|색상|품질|몇\s?(장|개)", "PRODUCT_INFO"),
    (r"매장|오프라인|지점|영업\s?시간|주차", "OTHER"),
]

CONF_THRESHOLD = 0.5


def classify(state: RouterState) -> RouterState:
    """노드 ① 분류 — 키워드 규칙 버전. 섹션 6에서 LLM 버전으로 갈아 끼운다."""
    q = state["question"]
    for pattern, route in RULES:
        if re.search(pattern, q):
            return {"route": route, "confidence": 0.8, "reason": "키워드 규칙 매치"}
    return {"route": "OTHER", "confidence": 0.3, "reason": "매치되는 규칙 없음"}


def gate(state: RouterState) -> RouterState:
    """노드 ② 판정 — 확신도와 응대 범위를 보고 처리/이관/범위밖을 정한다."""
    if state["confidence"] < CONF_THRESHOLD:
        return {"action": "ESCALATE",
                "message": "정확한 확인을 위해 상담원에게 연결해 드리겠습니다."}
    if state["route"] == "OTHER":
        return {"action": "OUT_OF_SCOPE",
                "message": "해당 내용은 주문하신 사이트의 고객센터를 통해 문의해 주셔야 확인이 가능합니다."}
    return {"action": "HANDLE", "message": None}


def build(node=None):
    """노드를 **이름으로** 찾아 그래프를 만든다.

    globals()[...] 로 찾기 때문에, 같은 이름의 함수를 다시 정의한 뒤 build() 를 한 번 더
    부르면 그 노드만 갈아 끼워진다. 그래프 정의는 오늘 하루 이 셀 하나뿐이다.
    (비교할 때는 node= 로 분류 노드를 직접 건네줄 수도 있다.)
    """
    g = StateGraph(RouterState)
    g.add_node("classify", node or globals()["classify"])
    g.add_node("gate", globals()["gate"])
    g.add_edge(START, "classify")
    g.add_edge("classify", "gate")
    g.add_edge("gate", END)
    return g.compile()


rule_classify = classify        # 나중에 기준선을 다시 잴 때 쓰려고 이름을 하나 붙여 둔다
app = build()
show_graph(app)

# 평가셋 120건에 적용
eval_set["pred_rule"] = [app.invoke({"question": q})["route"] for q in eval_set["question"]]

hit_mask = eval_set["pred_rule"] == eval_set["route"]
print(f"\n규칙 라우터 정확도: {100 * hit_mask.mean():.1f}% ({hit_mask.sum()}/{len(eval_set)})")

print("\n[라우트별 정답률]")
for r in ["ORDER_PLACE", "PRODUCT_INFO", "SHIPPING", "RETURN_REFUND"]:
    sub_df = eval_set[eval_set["route"] == r]
    hit = (sub_df["pred_rule"] == r).sum()
    print(f"  {r:15s} {hit:3d}/{len(sub_df):3d}  ({100 * hit / len(sub_df):5.1f}%)")
```

## 결과를 읽는다 — 어디가 무너졌는가

라우트별 정답률을 보면 편차가 큽니다. `RETURN_REFUND`와 `SHIPPING`은 비교적 잘 맞힙니다. "환불", "배송비"처럼 **그 의도에만 쓰이는 전용 어휘**가 존재하기 때문입니다. 반면 `ORDER_PLACE`는 처참합니다. 왜 그런가.

"낱개로도 구매 가능한가요?"라는 문장을 봅시다. 우리 규칙에서는 `낱개`와 `구매\s?가능`이 모두 `ORDER_PLACE` 규칙에 걸립니다. 그런데 정답셋은 이 문장을 `PRODUCT_INFO`로 라벨링한 경우가 있습니다. 고객이 실제로 원하는 것은 "낱개 판매를 하는지"라는 **상품 구성 정보**이기 때문이에요. 반대로 "이 반지 남성용만 따로 구매 가능한지 알려주세요"는 `ORDER_PLACE`입니다. 두 문장은 어휘가 거의 같아요.

이 지점이 오늘 수업의 핵심입니다. **키워드는 표면이고, 라우팅은 의도에 관한 판단입니다.** 매뉴얼 §1.1이 "고객이 쓴 단어가 아니라 고객이 원하는 결과로 판단한다"고 쓴 이유가 여기서 숫자로 확인됩니다. 어휘 규칙은 표면만 볼 수 있으므로 원리적으로 이 구분을 할 수 없습니다.

또 하나 관찰할 것이 있습니다. **순서가 결과를 바꾼다**는 점입니다. 우리 규칙은 위에서부터 검사하므로 "반품 배송비 얼마예요?"는 첫 규칙(`반품`)에 걸려 RETURN_REFUND가 됩니다. 만약 배송 규칙을 위에 두면 같은 문장이 `SHIPPING`이 됩니다. 어느 쪽이 맞을까. 매뉴얼 §1.2는 "진행 중인 반품·교환 건 문의는 교환·반품·환불로 분류한다"고 우선순위를 정해 두었습니다. 즉 **규칙의 순서조차 회사 문서에 근거가 있어야 합니다.**

➡️**Q. 규칙 순서와 키워드를 바꿔 실험하기**
```python
# Q. 규칙을 직접 고쳐 성능을 올려 보세요.
#    1) RULES의 순서를 바꾸면 점수가 어떻게 변하나요?
#    2) 새 키워드를 추가해 ORDER_PLACE 정답률을 올릴 수 있나요? 다른 라우트가 함께 떨어지지 않나요?
tests = [
    "반품 배송비 얼마예요?",          # 반품 + 배송비 — 두 규칙에 모두 걸린다
    "낱개로도 구매 가능한가요?",       # 정보 질문인가, 구매 의사인가
    "이 세트에서 토너만 따로 구매할 수 있나요?",
    "홍대점 오늘 몇 시까지 하나요?",   # 응대 범위 밖
    "ㅇㅇ",                           # 극단적으로 짧은 발화
]
for t in tests:
    s = app.invoke({"question": t})
    print(f'{s["route"]:15s} {s["action"]:13s} conf={s["confidence"]:.2f}  <- {t}')
```

### 🚀 더 해보기 — 규칙 라우터를 끝까지 밀어붙이기

**1. 규칙에 가중치를 주는 방식으로 바꿔 보기**

- 접근 힌트: 라우트별로 매치된 키워드 개수를 세어 점수가 가장 높은 라우트를 고르는 방식(bag-of-keywords)으로 `rule_router`를 다시 작성해 봅니다. `collections.Counter`를 쓰면 짧게 구현됩니다
- 성공하면: '먼저 매치되는 규칙 채택' 방식보다 나아지는지 숫자로 확인할 수 있습니다. 대부분 조금 오르지만 ORDER_PLACE/PRODUCT_INFO 혼동은 여전히 남는다는 것을 직접 보게 됩니다

**2. 전통적인 머신러닝 분류기와 비교해 보기**

- 접근 힌트: `sklearn.feature_extraction.text.TfidfVectorizer`로 문장을 벡터화하고 `LogisticRegression`을 학습시켜 봅니다. 학습 데이터는 문의 데이터 193건에서 eval 120건을 제외하고 씁니다
- 성공하면: LLM을 쓰지 않고도 상당한 성능이 나온다는 것과, 그래도 경계 사례에서는 여전히 헷갈린다는 것을 동시에 확인할 수 있습니다. LLM이 필요한 지점이 어디인지 근거를 갖고 말할 수 있게 됩니다

**3. 규칙 라우터를 LLM 라우터의 폴백으로 붙이기**

- 접근 힌트: API 호출이 실패하거나 응답이 형식에 맞지 않을 때 `rule_router`를 대신 호출하는 래퍼 함수를 만듭니다. `try/except`와 타임아웃 처리를 함께 넣습니다
- 성공하면: 실서비스에서 LLM 장애 시에도 서비스가 완전히 멈추지 않는 구조를 경험할 수 있습니다. 이 패턴은 오늘 오후 LLM 라우터를 만든 뒤 바로 적용해 볼 수 있습니다
