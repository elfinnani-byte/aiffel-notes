## 왜 기준액이 카테고리마다 다른 것이 함정인가

출력된 표를 봅시다. 무료배송 기준액이 액세서리 20,000원부터 신발 100,000원까지 다섯 배 차이가 납니다. 화장품은 아예 무료배송 대상이 아닙니다.

만약 에이전트가 "무료배송은 40,000원 이상"이라고 하드코딩했다면 어떻게 되는가. 의류 문의에서는 우연히 맞고, 신발 문의에서는 60,000원을 덜 담게 만들고, 화장품 문의에서는 존재하지 않는 혜택을 약속합니다. **우연히 맞는 경우가 있다는 것이 이 오류의 가장 위험한 성질입니다.** 테스트를 몇 개만 하면 "잘 되는 것 같다"는 결론이 나오기 때문이거든요.

이 함정은 우리가 일부러 심은 것이 아니라 **현실을 반영한 것**입니다. 오늘 데이터가 본뜬 실제 상담 기록에서 무료배송 기준액은 서로 다른 10종으로 관측되었습니다. 상담사들도 상품에 따라 다른 금액을 안내하고 있었다는 뜻입니다. 그래서 매뉴얼은 이 값을 본문에 적지 않고 조회 대상으로 지정했습니다.

이제 조회 함수를 만듭니다. 구현하면서 두 가지 원칙을 지킵니다.

**첫째, 없는 것은 없다고 답합니다.** 존재하지 않는 상품 ID를 받으면 빈 값이나 추측을 돌려주지 않고 명확한 오류를 돌려줍니다. 도구가 애매한 값을 돌려주면 모델이 그것을 채워 넣으려 합니다.

**둘째, `$`로 시작하는 키는 제거합니다.** 목 데이터에는 `$teaching_note` 같은 교육용 주석이 들어 있습니다. 이 주석에는 "귀책을 단정하면 안 됨" 같은 **정답에 해당하는 설명**이 적혀 있습니다. 이것이 그대로 모델에게 전달되면 모델은 스스로 판단한 것이 아니라 주석을 읽고 답한 것이 되어, 평가가 무의미해집니다. 실무에서도 내부 메모나 담당자 코멘트가 API 응답에 섞여 나가는 일이 있으므로, 응답을 정제하는 습관이 중요합니다.

➡️**실습: 조회 함수 ① 주문·상품**
```python
def clean(obj):
    """$ 로 시작하는 내부 주석 키를 재귀적으로 제거한다."""
    if isinstance(obj, dict):
        return {k: clean(v) for k, v in obj.items() if not k.startswith("$")}
    if isinstance(obj, list):
        return [clean(v) for v in obj]
    return obj

PRODUCTS = {p["product_id"]: p for p in MOCK["products"]}
ORDERS = {o["order_id"]: o for o in MOCK["orders"]}
RETURNS = {r["return_id"]: r for r in MOCK["returns"]}
RETURNS_BY_ORDER = {r["order_id"]: r for r in MOCK["returns"]}
RESTOCK = {r["product_id"]: r for r in MOCK["restock"]}
CATEGORIES = MOCK["categories"]

def search_product(query: str) -> dict:
    """상품명 일부로 상품을 찾는다. 상품 ID를 모를 때 가장 먼저 부르는 도구다.

    후보를 점수와 함께 돌려준다. 후보가 여럿이면 확정하지 말고 고객에게 되물어야 한다.
    """
    def toks(s):
        return [t for t in re.split(r"[\s·()]+", s) if t]

    qt = toks(query)
    if not qt:
        return {"query": query, "candidates": []}
    hits = []
    for pid, p in PRODUCTS.items():
        name, flat = p["name"], p["name"].replace(" ", "")
        nt = toks(name)
        n = sum(1 for t in qt if t in flat or any(t in x or x in t for x in nt))
        if n:
            hits.append({"product_id": pid, "name": name, "category": p["category"],
                         "price": p["price"], "score": round(n / len(qt), 2)})
    hits.sort(key=lambda h: -h["score"])
    top = [h for h in hits if h["score"] == hits[0]["score"]] if hits else []
    return {"query": query, "candidates": hits[:5],
            "resolved_product_id": top[0]["product_id"] if len(top) == 1 else None,
            "ambiguous": len(top) > 1,
            "note": ("후보가 여러 개입니다. 어느 상품인지 고객에게 확인하십시오."
                     if len(top) > 1 else None)}


def get_order_status(order_id: str) -> dict:
    """주문번호로 주문의 현재 진행 단계와 배송 정보를 조회한다."""
    o = ORDERS.get(order_id)
    if not o:
        return {"error": "주문을 찾을 수 없습니다", "order_id": order_id}
    return clean({k: o[k] for k in ["order_id", "status", "status_detail", "is_external_channel",
                                    "items", "order_amount", "shipping_fee", "address_region",
                                    "courier", "tracking_no", "invoice_printed",
                                    "expected_ship_date"] if k in o})

def get_product_detail(product_id: str) -> dict:
    """상품 ID로 구성·소재·원산지·재고·보증서 동봉 여부를 조회한다."""
    p = PRODUCTS.get(product_id)
    if not p:
        return {"error": "상품을 찾을 수 없습니다", "product_id": product_id}
    out = {k: p[k] for k in ["product_id", "name", "category", "price", "stock", "components",
                             "material", "origin", "has_quality_cert", "made_to_order",
                             "size_chart"] if k in p}
    out["category_label"] = CATEGORIES[p["category"]]["label"]
    return clean(out)

def get_product_options(product_id: str) -> dict:
    """상품 ID로 개별(단품) 구매 가능 여부와 판매 옵션을 조회한다."""
    p = PRODUCTS.get(product_id)
    if not p:
        return {"error": "상품을 찾을 수 없습니다", "product_id": product_id}
    return clean({k: p[k] for k in ["product_id", "name", "is_set", "components", "options",
                                    "individual_purchase_allowed", "individual_purchase_note",
                                    "made_to_order"] if k in p})

print(get_product_detail("P4001"))
print()
print(get_product_options("P1001"))
```

## 배송정책 조회 — 계산까지 도구가 맡는다

다음은 오늘의 함정과 직접 맞붙는 함수입니다. `get_shipping_policy`는 상품 ID로 카테고리를 찾아 그 카테고리의 무료배송 기준액을 돌려줍니다. 여기에 하나를 더 합니다. **결제 금액을 함께 받으면 부족액까지 계산해서 돌려줍니다.**

왜 계산까지 도구가 하는가. 모델에게 "기준 100,000원, 현재 59,000원"만 주고 부족액을 계산하게 하면 대개 맞히지만 틀리기도 합니다. 언어 모델의 산술은 신뢰할 수 있는 기능이 아닙니다. 그리고 뒤에서 만들 가드레일은 답변 속 숫자의 출처를 조회 결과에서 찾는데, 모델이 계산한 41,000원은 조회 결과에 없으므로 출처 불명으로 걸립니다. **계산이 필요하면 코드가 계산해서 도구 응답에 담는 것**이 정답이죠.

실제 어드민 화면도 이렇게 동작합니다. 상담원이 주문 금액을 넣으면 화면이 부족액을 표시해 줍니다. 사람이 암산하게 만들지 않습니다.

➡️**실습: 조회 함수 ② 배송정책과 부족액 계산**
```python
# 선택 인자는 Optional 로 적는다. 타입이 int 인데 기본값이 None 이면
# 모델이 '모름'의 뜻으로 None 을 넣었을 때 스키마 검증에서 거부당한다.
from typing import Optional

# 목 데이터의 지역 키는 '수도권 / 수도권외 / 제주도서산간' 세 가지뿐이다.
# 고객은 "제주도", "부산"처럼 말하므로 그 사이를 메워 주는 정규화가 필요하다.
CAPITAL = ("서울", "경기", "인천", "수도권")


def normalize_region(region):
    """고객이 말한 지역명을 목 데이터의 지역 키로 옮긴다."""
    if not region:
        return None
    if any(k in region for k in ("제주", "도서", "산간", "울릉")):
        return "제주도서산간"
    if any(k in region for k in CAPITAL):
        return "수도권"
    return "수도권외"


def region_info(region):
    """지역별 당일배송 제공 여부와 도서산간 추가 배송비를 돌려준다."""
    key = normalize_region(region)
    sd = clean(MOCK["same_day_delivery"].get(key, {}))
    return {"region_input": region, "region": key,
            "same_day_available": sd.get("available", False),
            "same_day_fee": sd.get("fee"),
            "extra_shipping_fee": sd.get("extra_fee"),     # 도서산간 추가 배송비
            "region_note": sd.get("note")}


def get_shipping_policy(product_id: Optional[str] = None, category: Optional[str] = None,
                        order_amount: Optional[int] = None,
                        region: Optional[str] = None) -> dict:
    """무료배송 기준액과 배송비를 조회한다. 금액을 주면 부족액까지 계산한다.

    상품 ID 로도, 카테고리로도 조회할 수 있다. 지역만 궁금한 문의(도서산간 추가비 등)는
    상품 없이 region 만 줘도 된다. region 을 주면 당일배송 제공 여부도 함께 돌려준다.
    """
    p = PRODUCTS.get(product_id) if product_id else None
    if product_id and not p:
        return {"error": "상품을 찾을 수 없습니다", "product_id": product_id}
    cat_key = p["category"] if p else category
    if cat_key is None:                      # 상품도 카테고리도 없다 — 지역 정보만 돌려준다
        out = {"base_shipping_fee": FIXED_POLICY["base_shipping_fee"],
               "note": "무료배송 기준은 카테고리마다 다릅니다. 상품 또는 카테고리를 지정해 주세요."}
        if region:
            out.update(region_info(region))
        return clean(out)
    if cat_key not in CATEGORIES:
        return {"error": "카테고리를 찾을 수 없습니다", "category": cat_key}
    cat = CATEGORIES[cat_key]
    base = FIXED_POLICY["base_shipping_fee"]
    th = cat["free_shipping_threshold"]

    out = {"product_id": product_id, "name": p["name"] if p else None, "category": cat_key,
           "category_label": cat["label"], "price": p["price"] if p else None,
           "base_shipping_fee": base, "free_shipping_threshold": th}
    if th is None:
        out["free_shipping_available"] = False
        out["note"] = cat.get("free_shipping_note", "무료배송 대상이 아닙니다.")
    else:
        out["free_shipping_available"] = True

    if order_amount is not None:
        out["order_amount"] = order_amount
        if th is None:
            out["free_shipping_applied"] = False
            out["shipping_fee"] = base
        elif order_amount >= th:
            out["free_shipping_applied"] = True
            out["shipping_fee"] = 0
        else:
            out["free_shipping_applied"] = False
            out["shipping_fee"] = base
            out["shortfall"] = th - order_amount        # 부족액은 코드가 계산한다

    if region:
        out.update(region_info(region))
    return clean(out)

print("[제주도 배송비 — 상품 없이 지역만]")
print(get_shipping_policy(region="제주도"))
print("\n[신발 P4001 · 결제 59,000원]")
print(get_shipping_policy("P4001", 59000))
print("\n[화장품 P5001 · 결제 50,000원] — 금액과 무관하게 무료배송 대상 아님")
print(get_shipping_policy("P5001", 50000))
print("\n[가방·잡화 P6002 · 결제 78,000원] — 기준 50,000원 충족")
print(get_shipping_policy("P6002", 78000))
```

## 같은 질문에 다른 답이 나온다는 것을 눈으로 확인했다

세 상품에 같은 질문("무료배송 되나요?")을 던졌는데 답이 전부 다릅니다. 신발은 100,000원 기준에 41,000원이 부족하고, 화장품은 기준 자체가 없어 얼마를 사도 배송비가 붙고, 가방은 기준을 충족해 배송비가 0원이에요.

이것이 조회가 필요한 이유를 가장 분명하게 보여주는 장면입니다. **문의 문장만 보면 세 경우가 완전히 같습니다.** 다른 것은 상품입니다. 그러므로 상품을 특정하지 않으면 답할 수 없고, 상품을 특정했다면 반드시 조회해야 합니다.

여기서 자연스럽게 따라오는 규칙이 하나 있습니다. 고객이 상품을 특정하지 않고 "배송비 얼마예요?"라고 물으면 에이전트는 **되물어야 합니다.** 매뉴얼 §4.2가 "상품명을 알려 주시면 정확히 확인해 드리겠습니다"라는 응대를 규정한 이유입니다. 이때 기본 배송비 2,500원은 전 상품 공통이므로 먼저 안내해도 됩니다. **고정값은 답하고 변동값은 되묻는** 이 조합이 오늘 배우는 응대의 기본형이에요.

남은 조회 함수들을 완성합시다. 반품 관련 조회에는 특별히 조심할 점이 있는데, 곧 설명합니다.

➡️**실습: 조회 함수 ③ 반품·재입고·이관**
```python
# 선택 인자는 Optional 로 적는다. 타입이 int 인데 기본값이 None 이면
# 모델이 '모름'의 뜻으로 None 을 넣었을 때 스키마 검증에서 거부당한다.
from typing import Optional

def get_return_policy(product_id: Optional[str] = None, order_id: Optional[str] = None) -> dict:
    """상품 또는 주문의 반품 가능 기간과 조건을 조회한다."""
    if order_id:
        o = ORDERS.get(order_id)
        if not o:
            return {"error": "주문을 찾을 수 없습니다", "order_id": order_id}
        product_id = o["items"][0]["product_id"]
    p = PRODUCTS.get(product_id)
    if not p:
        return {"error": "상품을 찾을 수 없습니다", "product_id": product_id}
    cat = CATEGORIES[p["category"]]
    return clean({
        "product_id": product_id, "name": p["name"], "category": p["category"],
        "category_label": cat["label"],
        "return_window_days": cat["return_window_days"],
        "return_window_basis": cat["return_window_basis"],
        "requires_unopened": cat["requires_unopened"],
        "made_to_order": p.get("made_to_order", False),
        "return_blocked": bool(p.get("made_to_order")),
        "return_blocked_reason": "주문 제작 상품은 교환·반품이 불가합니다." if p.get("made_to_order") else None,
    })

def get_return_status(order_id: Optional[str] = None, return_id: Optional[str] = None) -> dict:
    """반품·교환의 현재 처리 단계를 조회한다. 검품 전에는 귀책이 확정되지 않는다."""
    r = RETURNS.get(return_id) if return_id else RETURNS_BY_ORDER.get(order_id)
    if not r:
        return {"error": "반품 접수 내역을 찾을 수 없습니다",
                "order_id": order_id, "return_id": return_id}
    return clean(r)

def get_restock_info(product_id: str) -> dict:
    """품절 상품의 재입고 확정 여부와 예정일을 조회한다."""
    r = RESTOCK.get(product_id)
    if not r:
        p = PRODUCTS.get(product_id)
        if p and p.get("stock"):
            return {"product_id": product_id, "is_soldout": False,
                    "stock": p["stock"], "note": "재고가 있어 재입고 대기 상품이 아닙니다."}
        return {"error": "재입고 정보를 찾을 수 없습니다", "product_id": product_id}
    return clean(r)

def escalate_to_agent(reason: str, context: Optional[dict] = None) -> dict:
    """상담원에게 이관한다. 매뉴얼 7.2의 이관 기준에 해당할 때 호출한다."""
    return {"escalated": True, "reason": reason, "context": context or {},
```

## `null` 이 가장 중요한 값이다

`R-2001`의 조회 결과를 봅시다. `inspection_result`, `fault_party`, `shipping_fee_bearer`가 모두 `None`(JSON의 `null`)입니다. 이것은 **데이터가 없어서 비어 있는 것이 아니라, 아직 결정되지 않았다는 정보**입니다. 검품이 끝나지 않았으므로 배송비를 누가 부담하는지 확정할 수 없는 상태예요.

이 구분이 실무에서 엄청나게 중요합니다. `None`을 "값 없음"으로 취급해 무시하면 에이전트는 다른 근거를 찾아 답을 만들려 하고, 결국 "고객님이 부담하셔야 합니다" 같은 단정을 하게 됩니다. 그런데 검품 결과가 하자로 나오면 그 안내는 틀린 것이 되고, 이미 한 안내를 번복해야 합니다. 매뉴얼 §6.4가 이 지점을 "항의가 가장 많이 발생하는 지점"으로 명시하고, 접수 시에는 모든 고객에게 배송비 발생 가능성을 안내하되 검품 후 하자로 확인되면 차감 없이 환불한다고 규정한 이유이죠.

그러므로 에이전트는 `None`을 만나면 **"아직 확정되지 않았다"를 답변에 그대로 반영**해야 합니다. 이것은 도구 응답의 형태만으로는 강제되지 않으므로, 프롬프트와 가드레일로 함께 다뤄야 합니다. 오늘 오후에 정확히 그 작업을 합니다.

재입고 정보도 같은 구조입니다. `P4002`는 `is_confirmed`가 `false`이므로 예정일을 확답하면 안 됩니다. 반면 목 데이터에는 `is_confirmed`가 `true`인 상품도 있어서, 그 경우에는 예정일을 안내해도 됩니다. **같은 종류의 문의인데 데이터 상태에 따라 답이 반대가 되는 것**입니다. 정답셋은 이 두 경우를 짝으로 갖추고 있고, 오후에 채점해 보면 우리 에이전트가 그 차이를 구분하는지 드러납니다.

> [!question]+ 객관식 퀴즈
> **Q. 반품 조회 결과에서 `inspection_result`가 `null`이었습니다. 에이전트가 취해야 할 올바른 행동은 무엇인가요?**
>
> 1. 검품이 끝나지 않아 귀책이 미확정임을 밝히고, 모든 반품 건에 동일하게 적용되는 절차임을 설명합니다
> 2. 고객이 하자라고 말했으므로 하자로 간주해 배송비 무료를 안내합니다
> 3. 값이 없으므로 매뉴얼의 단순 변심 기준을 적용해 고객 부담으로 안내합니다
> 4. null은 오류이므로 조회를 다시 시도하고, 계속 null이면 임의로 판단합니다
>
> ![힌트](https://learn.modulabs.co.kr/_next/static/media/Thick.72cfead3.svg)힌트
> >정답은 1번입니다. null은 '데이터 없음'이 아니라 '아직 확정되지 않음'이라는 상태 정보이며, 매뉴얼 §6.4는 하자 여부가 입고 후 검품 단계에서 확정된다고 규정합니다. 2번과 3번은 방향만 반대인 같은 오류로, 확정되지 않은 것을 단정하는 것입니다. 특히 3번처럼 고객 부담으로 단정하면 나중에 하자로 판정될 때 번복해야 하고, 이는 매뉴얼이 지적한 최대 항의 지점을 건드립니다. 4번은 재시도로 해결되는 문제가 아니며, '임의로 판단'은 어떤 경우에도 허용되지 않습니다.
